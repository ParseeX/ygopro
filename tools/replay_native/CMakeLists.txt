cmake_minimum_required(VERSION 3.16)
project(replay_native LANGUAGES C CXX)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CORE_SOURCE "${REPO_ROOT}/ocgcore" CACHE PATH "与录像匹配的核心源码")
set(LUA_SOURCE "${REPO_ROOT}/lua/src" CACHE PATH "Lua 源码目录")
file(GLOB LUA_FILES CONFIGURE_DEPENDS "${LUA_SOURCE}/*.c")
list(FILTER LUA_FILES EXCLUDE REGEX "/(lua|luac)\\.c$")
set_source_files_properties(${LUA_FILES} PROPERTIES LANGUAGE CXX)
add_library(replay_lua STATIC ${LUA_FILES})
set_target_properties(replay_lua PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(replay_lua PUBLIC "${LUA_SOURCE}")
file(GLOB CORE_FILES CONFIGURE_DEPENDS "${CORE_SOURCE}/*.cpp")
add_library(replay_core SHARED bridge.cpp ${CORE_FILES})
target_compile_definitions(replay_core PRIVATE OCGCORE_EXPORT_FUNCTIONS)
target_compile_features(replay_core PRIVATE cxx_std_17)
target_include_directories(replay_core PRIVATE "${CORE_SOURCE}")
target_link_libraries(replay_core PRIVATE replay_lua)
if(MINGW)
  target_link_options(replay_core PRIVATE -static-libgcc -static-libstdc++ -static)
elseif(UNIX)
  target_link_libraries(replay_core PRIVATE m ${CMAKE_DL_LIBS})
endif()
